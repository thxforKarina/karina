        <source src="Karina.mp4" type="video/mp4">
    </video>
    <button id="playButton">▶ СМОТРЕТЬ</button>
    <script>
        const video=document.getElementById('myVideo');
        const playButton=document.getElementById('playButton');
        function playVideoWithSound(){
            video.muted=false;
            video.play().then(()=>{
                playButton.style.display='none';
            }).catch(()=>{
                video.muted=true;
                video.play().then(()=>{
                    playButton.style.display='block';
                }).catch(()=>{
                    playButton.style.display='block';
                });
            });
        }
        if(video.paused){
            playVideoWithSound();
        }else{
            video.muted=false;
            if(video.paused){
                video.muted=true;
                video.play();
                playButton.style.display='block';
            }
        }
        playButton.addEventListener('click',function(){
            video.muted=false;
            video.play().then(()=>{
                playButton.style.display='none';
            }).catch(()=>{
                video.muted=true;
                video.play();
                playButton.style.display='none';
            });
        });
        document.body.addEventListener('click',function(){
            if(video.paused){
                video.muted=false;
                video.play().catch(()=>{
                    video.muted=true;
                    video.play();
                });
                playButton.style.display='none';
            }
        });
    </script>
